#   pip install --upgrade pip
#
# [Step 3] 나머지 패키지를 인터넷 머신에서 미리 다운로드

#   pip download \
#     --only-binary=:all: \
#     --platform manylinux_2_28_x86_64 \
#     --python-version 39 \
#     --implementation cp \
#     --abi cp39 \
#     -r requirements.txt \
#     -d ./wheels/
#   tar czf aml-wheels.tar.gz wheels/
#   scp aml-wheels.tar.gz systest@<클러스터-호스트>:/tmp/
#
# [Step 4] 클러스터 노드에서 오프라인 설치
#   cd /tmp && tar xzf aml-wheels.tar.gz
#   source /tmp/aml-venv/bin/activate
#   pip install --no-index --find-links=./wheels/ -r requirements.txt
#
# [Step 5] 설치 확인
#   python3 -c "import gssapi, kafka, sdv; print('OK')"
#
